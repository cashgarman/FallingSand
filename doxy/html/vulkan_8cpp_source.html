20// Vulkan device, swapchain, and compute passes. This is the private side of the pimpl.
21//
22// Owner is a move-only RAII handle. A concept requires the destroy function to be
23// callable with the parent, the handle, and the allocator callbacks. Copy is deleted
24// and move uses exchange so the source keeps an empty handle. The five special
25// members agree on that single owner.
26//
27// source_location records the call site of a failed Vulkan result, and format builds
28// the exception text. Atomics use relaxed ordering for a validation-error counter
29// that does not publish other data. Designated initializers name the Vulkan fields
30// each call sets. Ranges search extension and format lists in place. string_view
31// compares extension names without allocating. enumerate pairs a frame slot with
32// its index. countr_zero turns a composite-alpha bit mask into a bit index.
33// filesystem::path finds the shader directory beside the executable.
34//
35// Returning a move-only Owner by value lets the compiler move it into the caller.
36// std::move transfers a replacement swapchain into the one the frame loop keeps.
37// memcpy copies trivially copyable cells and quads. The public destructor is
38// defaulted in this file because unique_ptr needs Impl to be complete.

64// Parent is the Vulkan object the destroy function needs. Handle is the handle type.
65// Destroy is a compile-time function pointer, so each handle kind gets its own Owner.
66// The requires-clause drops the template unless Destroy can be called with those arguments.
67template <class Parent, class Handle, auto Destroy>
68 requires std::invocable<decltype(Destroy), Parent, Handle, const VkAllocationCallbacks*>
69class Owner
70{
71 public:
72
73 // A defaulted constructor owns nothing until put() acquires a handle.
74 Owner() = default;
75
76 ~Owner()
77 {
78 reset();
79 }
80
81 // Deleted copy. Two owners would pass the same handle to Destroy.
82 Owner(const Owner&) = delete;
83 Owner& operator=(const Owner&) = delete;
84
85 // Move takes the handle and is noexcept because copying two handles cannot fail.
86 // exchange leaves the source with an empty value.
87 Owner(Owner&& other) noexcept : parent_(other.parent_), value_(std::exchange(other.value_, {}))
88 {
89 }
90
91 Owner& operator=(Owner&& other) noexcept
92 {
93 // Skip self-move. Releasing first would destroy the handle this object still needs.
94 if (this != &other)
95 {
96 reset();
97 parent_ = other.parent_;
98
99 // Install an empty handle in the source and keep the previous value here.
100 value_ = std::exchange(other.value_, {});
101 }
102 return *this;
103 }
104
105 [[nodiscard]] Handle get() const
106 {
107 return value_;
108 }
109
110 Handle* put(Parent parent)
111 {
112 reset();
113 parent_ = parent;
114 return &value_;
115 }
116
117 void reset()
118 {
119 if (value_)
120 {
121 // exchange returns the live handle and stores an empty one before Destroy runs.
122 Destroy(parent_, std::exchange(value_, {}), nullptr);
123 }
124 }
125
126 private:
127 Parent parent_{};
128 Handle value_{};
129};

376 {
377 // Record the device, then put() the new buffer and memory into their Owners.
378 device = ctx.device;

385 };
386 check(vkCreateBuffer(device, &info, nullptr, buffer.put(device)));
387 VkMemoryRequirements requirements{};
388 vkGetBufferMemoryRequirements(device, buffer.get(), &requirements);
389 const VkMemoryPropertyFlags flags =
390 host ? VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT | VK_MEMORY_PROPERTY_HOST_COHERENT_BIT

393 for (; i < ctx.memory.memoryTypeCount; ++i)
394 {
395 if ((requirements.memoryTypeBits & (1u << i)) != 0 &&
396 (ctx.memory.memoryTypes[i].propertyFlags & flags) == flags)
397 {
398 break;
399 }
400 }
401 if (i == ctx.memory.memoryTypeCount)
402 {
403 throw std::runtime_error("Vulkan memory type unavailable.");
404 }
405 const VkMemoryAllocateInfo allocation{
406 .sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO,
407 .allocationSize = requirements.size,
408 .memoryTypeIndex = i,
409 };
410 check(vkAllocateMemory(device, &allocation, nullptr, memory.put(device)));
411 check(vkBindBufferMemory(device, buffer.get(), memory.get(), 0));
412 if (host)

580void dumpBitmap(const std::filesystem::path& path, const Buffer& source, VkExtent2D extent,
581 VkFormat format)
582{
583 std::array<unsigned char, 54> header{};
584
585 // The lambda captures the header by reference and writes one little-endian word.
586 const auto set = [&header](std::size_t offset, std::uint32_t value) {
587 for (unsigned i = 0; i < 4; ++i)
588 {
589 header[offset + i] = static_cast<unsigned char>(value >> (8 * i));
590 }
591 };
592 header[0] = 'B';
593 header[1] = 'M';
594 set(2, static_cast<std::uint32_t>(header.size() + source.size));
595 set(10, static_cast<std::uint32_t>(header.size()));
596 set(14, 40);
597 set(18, extent.width);
598 set(22, 0u - extent.height);
599 header[26] = 1;
600 header[28] = 32;
601 set(34, static_cast<std::uint32_t>(source.size));
602
603 const auto* data = static_cast<const char*>(source.mapped);
604 std::vector<char> swizzled;
605 if (format == VK_FORMAT_R8G8B8A8_SRGB)
606 {
607 swizzled.assign(data, data + source.size);
608 for (std::size_t i = 0; i < swizzled.size(); i += 4)
609 {
610 std::swap(swizzled[i], swizzled[i + 2]);
611 }
612 data = swizzled.data();
613 }
614
615 // ofstream owns the screenshot file and closes it when this function returns.
616 std::ofstream output(path, std::ios::binary);
617 output.write(reinterpret_cast<const char*>(header.data()), header.size());
618 output.write(data, static_cast<std::streamsize>(source.size));
619 output.close();
620 if (!output)
621 {
622 throw std::runtime_error("Could not save screenshot.");
623 }
624}

682 std::array<VkCommandBuffer, framesInFlight + 1> buffers{};
683 const VkCommandBufferAllocateInfo allocate{
684 .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO,
685 .commandPool = commands.get(),
686 .level = VK_COMMAND_BUFFER_LEVEL_PRIMARY,
687 .commandBufferCount = window ? static_cast<std::uint32_t>(buffers.size()) : 1u,
688 };
689 check(vkAllocateCommandBuffers(ctx.device, &allocate, buffers.data()));
690 immediate = buffers.front();
691 const VkFenceCreateInfo immediateInfo{.sType = VK_STRUCTURE_TYPE_FENCE_CREATE_INFO};
692 check(vkCreateFence(ctx.device, &immediateInfo, nullptr, immediateFence.put(ctx.device)));
693
694 constexpr auto usage = VK_BUFFER_USAGE_STORAGE_BUFFER_BIT | VK_BUFFER_USAGE_TRANSFER_SRC_BIT |
695 VK_BUFFER_USAGE_TRANSFER_DST_BIT;
696 // Allocate every simulation and auxiliary GPU buffer for this device.
697 for (std::size_t i = 0; i < storage.size(); ++i)
698 {
699 storage[i].create(ctx, bufferSizes[i], usage);
700 }
701 staging.create(ctx, std::max(cellBytes, flowBytes * 2),
702 VK_BUFFER_USAGE_TRANSFER_SRC_BIT | VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
703
704 if (window)
705 {
706 constexpr VkFenceCreateInfo fenceInfo{
707 .sType = VK_STRUCTURE_TYPE_FENCE_CREATE_INFO,
708 .flags = VK_FENCE_CREATE_SIGNALED_BIT,
709 };
710 constexpr VkSemaphoreCreateInfo semInfo{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO};
711
712 // enumerate names the slot index. The command buffer for that slot sits one past immediate.
713 for (auto [i, slot] : std::views::enumerate(slots))
714 {
715 slot.commands = buffers[static_cast<std::size_t>(i) + 1];
716 check(vkCreateFence(ctx.device, &fenceInfo, nullptr, slot.fence.put(ctx.device)));
717 check(vkCreateSemaphore(ctx.device, &semInfo, nullptr, slot.available.put(ctx.device)));
718 slot.ui.create(ctx, sizeof(Quad) * maxQuads, VK_BUFFER_USAGE_STORAGE_BUFFER_BIT, true);
719 }
720 if (ctx.timestampBits)
721 {
722 const VkQueryPoolCreateInfo query{
723 .sType = VK_STRUCTURE_TYPE_QUERY_POOL_CREATE_INFO,
724 .queryType = VK_QUERY_TYPE_TIMESTAMP,
725 .queryCount = 3 * framesInFlight,
726 };
727 check(vkCreateQueryPool(ctx.device, &query, nullptr, queries.put(ctx.device)));
728 }
729 }
730
731 setupDescriptors(window != nullptr);
732
733 // Build one compute pipeline per simulation kernel .spv module.
734 for (std::size_t i = 0; i < compute.size(); ++i)
735 {
736 auto module = shader(kernelNames[i]);
737 const VkComputePipelineCreateInfo info{
738 .sType = VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO,
739 .stage =
740 {
741 .sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
742 .stage = VK_SHADER_STAGE_COMPUTE_BIT,
743 .module = module.get(),
744 .pName = "main",
745 },
746 .layout = pipelineLayout.get(),
747 };
748 check(vkCreateComputePipelines(ctx.device, {}, 1, &info, nullptr,
749 compute[i].put(ctx.device)));
750 }
751}

838DeviceOwner<VkShaderModule, vkDestroyShaderModule> VulkanGame::Impl::shader(
839 std::string_view name) const
840{
841 // string_view borrows the shader name. format and path build the .spv location.
842 // ifstream owns the file, opened at the end so the size is one tellg.
843 const auto file = shaders / std::format("{}.spv", name);
844 std::ifstream input(file, std::ios::binary | std::ios::ate);
845 const auto end = input.tellg();
846 if (end <= 0 || end % 4 != 0)
847 {
848 throw std::runtime_error(std::format("Could not load shader {}.", file.string()));
849 }
850 std::vector<std::uint32_t> words(static_cast<std::size_t>(end) / sizeof(std::uint32_t));
851 input.seekg(0);
852 if (!input.read(reinterpret_cast<char*>(words.data()), end))
853 {
854 throw std::runtime_error(std::format("Could not read shader {}.", file.string()));
855 }
856 const VkShaderModuleCreateInfo info{
857 .sType = VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO,
858 .codeSize = words.size() * sizeof(std::uint32_t),
859 .pCode = words.data(),
860 };
861 DeviceOwner<VkShaderModule, vkDestroyShaderModule> module;
862 check(vkCreateShaderModule(ctx.device, &info, nullptr, module.put(ctx.device)));
863
864 // Return the named owner directly. It is move-only, so the caller receives the handle.
865 return module;

868void VulkanGame::Impl::resize(int width, int height)
869{
870 // Rebuild the swapchain after the device is idle. The old one is passed in as oldSwapchain.
871 check(vkDeviceWaitIdle(ctx.device));
872 VkSurfaceCapabilitiesKHR capabilities{};
873 check(vkGetPhysicalDeviceSurfaceCapabilitiesKHR(ctx.gpu, ctx.surface, &capabilities));
874 if (capabilities.currentExtent.width == 0 || capabilities.currentExtent.height == 0)
875 {
876 return;
877 }
878 std::uint32_t count{};
879 check(vkGetPhysicalDeviceSurfaceFormatsKHR(ctx.gpu, ctx.surface, &count, nullptr));
880 std::vector<VkSurfaceFormatKHR> formats(count);
881 check(vkGetPhysicalDeviceSurfaceFormatsKHR(ctx.gpu, ctx.surface, &count, formats.data()));
882
883 // find_if returns an iterator. The lambda tests one format and captures nothing.
884 auto srgb = std::ranges::find_if(formats, [](VkSurfaceFormatKHR f) {
885 return (f.format == VK_FORMAT_B8G8R8A8_SRGB || f.format == VK_FORMAT_R8G8B8A8_SRGB) &&
886 f.colorSpace == VK_COLOR_SPACE_SRGB_NONLINEAR_KHR;
887 });
888 if (srgb == formats.end() &&
889 !(formats.size() == 1 && formats.front().format == VK_FORMAT_UNDEFINED))
890 {
891 throw std::runtime_error("The surface does not provide an sRGB swapchain format.");
892 }
893
894 Swapchain replacement;
895 replacement.format = srgb == formats.end() ? VK_FORMAT_B8G8R8A8_SRGB : srgb->format;
896 replacement.extent = capabilities.currentExtent;
897 if (replacement.extent.width == std::numeric_limits<std::uint32_t>::max())
898 {
899 replacement.extent = {
900 std::clamp(static_cast<std::uint32_t>(width), capabilities.minImageExtent.width,
901 capabilities.maxImageExtent.width),
902 std::clamp(static_cast<std::uint32_t>(height), capabilities.minImageExtent.height,
903 capabilities.maxImageExtent.height),
904 };
905 }
906 auto presentMode = VK_PRESENT_MODE_FIFO_KHR;
907 check(vkGetPhysicalDeviceSurfacePresentModesKHR(ctx.gpu, ctx.surface, &count, nullptr));
908 std::vector<VkPresentModeKHR> modes(count);
909 check(vkGetPhysicalDeviceSurfacePresentModesKHR(ctx.gpu, ctx.surface, &count, modes.data()));
910 for (auto mode : {VK_PRESENT_MODE_IMMEDIATE_KHR, VK_PRESENT_MODE_MAILBOX_KHR})
911 {
912 // contains reports whether this present mode is in the list the surface advertised.
913 if (std::ranges::contains(modes, mode))
914 {
915 presentMode = mode;
916 break;
917 }
918 }
919 replacement.captureSupported =
920 (capabilities.supportedUsageFlags & VK_IMAGE_USAGE_TRANSFER_SRC_BIT) != 0;
921 const auto minImages =
922 capabilities.maxImageCount == 0
923 ? capabilities.minImageCount + 1
924 : std::min(capabilities.minImageCount + 1, capabilities.maxImageCount);
925
926 // countr_zero finds the lowest set bit in the mask, which is the alpha mode to request.
927 const auto alpha = static_cast<VkCompositeAlphaFlagBitsKHR>(
928 1u << std::countr_zero(capabilities.supportedCompositeAlpha));
929 const VkSwapchainCreateInfoKHR info{
930 .sType = VK_STRUCTURE_TYPE_SWAPCHAIN_CREATE_INFO_KHR,
931 .surface = ctx.surface,
932 .minImageCount = minImages,
933 .imageFormat = replacement.format,
934 .imageColorSpace = VK_COLOR_SPACE_SRGB_NONLINEAR_KHR,
935 .imageExtent = replacement.extent,
936 .imageArrayLayers = 1,
937 .imageUsage = VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT |
938 (replacement.captureSupported
939 ? static_cast<VkImageUsageFlags>(VK_IMAGE_USAGE_TRANSFER_SRC_BIT)
940 : 0u),
941 .imageSharingMode = VK_SHARING_MODE_EXCLUSIVE,
942 .preTransform = capabilities.currentTransform,
943 .compositeAlpha = alpha,
944 .presentMode = presentMode,
945 .clipped = VK_TRUE,
946 .oldSwapchain = swap.handle.get(),
947 };
948 check(vkCreateSwapchainKHR(ctx.device, &info, nullptr, replacement.handle.put(ctx.device)));
949 check(vkGetSwapchainImagesKHR(ctx.device, replacement.handle.get(), &count, nullptr));
950 replacement.images.resize(count);
951 replacement.views.resize(count);
952 replacement.finished.resize(count);
953 check(vkGetSwapchainImagesKHR(ctx.device, replacement.handle.get(), &count,
954 replacement.images.data()));
955 const VkSemaphoreCreateInfo semaphore{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO};
956 for (std::uint32_t i = 0; i < count; ++i)
957 {
958 const VkImageViewCreateInfo view{
959 .sType = VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO,
960 .image = replacement.images[i],
961 .viewType = VK_IMAGE_VIEW_TYPE_2D,
962 .format = replacement.format,
963 .subresourceRange = {VK_IMAGE_ASPECT_COLOR_BIT, 0, 1, 0, 1},
964 };
965 check(vkCreateImageView(ctx.device, &view, nullptr, replacement.views[i].put(ctx.device)));
966 check(vkCreateSemaphore(ctx.device, &semaphore, nullptr,
967 replacement.finished[i].put(ctx.device)));
968 }
969 const bool changed = swap.format != replacement.format;
970
971 // Drop the old views and semaphores before the move, then transfer the new swapchain.
972 swap.views.clear();
973 swap.finished.clear();
974 swap = std::move(replacement);
975 if (changed)
976 {
977 makeGraphics();
978 }
979 outdated = false;
980}

1086void VulkanGame::Impl::recordBrush(VkCommandBuffer cmd, const Stroke& stroke)
1087{
1088 push.operation = {stroke.stir ? 1u : 0u, 0, std::to_underlying(stroke.material), 0};
1089 push.brush = {stroke.from.x, stroke.from.y, stroke.to.x, stroke.to.y};
1090 push.cursor[2] = stroke.radius;
1091
1092 // Init-capture copies the radius into the lambda. The stroke need not stay alive later.
1093 const auto covering = [radius = stroke.radius](float a, float b) {
1094 return static_cast<unsigned>(std::ceil(std::abs(a - b) + 2 * radius)) + 2;
1095 };
1096 dispatch(cmd, Kernel::brush, groups(covering(stroke.from.x, stroke.to.x), 16),
1097 groups(covering(stroke.from.y, stroke.to.y), 8));
1098}

1100void VulkanGame::Impl::recordSim(VkCommandBuffer cmd, unsigned ticks)
1101{
1102 // One ordinary loop per tick. Each dispatch finishes recording before the next is written.
1103 constexpr unsigned fullX = groups(worldWidth, 16), fullY = groups(worldHeight, 8);
1104 for (unsigned step = 0; step < ticks; ++step)
1105 {
1106 ++push.tick;
1107 dispatch(cmd, Kernel::reaction, fullX, fullY);
1108 push.state ^= 1u;
1109 dispatch(cmd, Kernel::grains, groups((worldWidth + 2) / 2, 16),
1110 groups((worldHeight + 2) / 2, 8));
1111 dispatch(cmd, Kernel::predict, fullX, fullY);
1112 dispatch(cmd, Kernel::equation, fullX, fullY);
1113 // Red-black pressure solves alternate parity each pass.
1114 for (unsigned pass = 0; pass < pressureBatches; ++pass)
1115 {
1116 push.operation[1] = pass & 1u;
1117 dispatch(cmd, Kernel::pressure, groups(worldWidth, 12), groups(worldHeight, 12));
1118 }
1119 dispatch(cmd, Kernel::project, fullX, fullY);
1120 dispatch(cmd, Kernel::flux, fullX, fullY);
1121 // Flux limiting runs an odd number of passes so the field settles.
1122 for (unsigned pass = 0; pass < fluxLimitPasses; ++pass)
1123 {
1124 push.operation = {pass, pass & 1u, 0, 0};
1125 dispatch(cmd, Kernel::limit, fullX, fullY);
1126 }
1127 dispatch(cmd, Kernel::advect, fullX, fullY);
1128 push.state ^= 1u;
1129
1130 // Separate liquid phases on two checkerboard passes.
1131 for (unsigned phase = 0; phase < 2; ++phase)
1132 {
1133 push.operation[1] = (phase + push.tick) & 1u;
1134 dispatch(cmd, Kernel::separate, fullX, groups(worldHeight / 2, 8));
1135 }
1136 }

1139void VulkanGame::Impl::readTiming(const FrameSlot& slot, std::uint32_t first)
1140{
1141 if (!queries.get() || !slot.measured)
1142 {
1143 return;
1144 }
1145 std::array<std::uint64_t, 3> stamps{};
1146 check(vkGetQueryPoolResults(ctx.device, queries.get(), first,
1147 static_cast<std::uint32_t>(stamps.size()), sizeof(stamps),
1148 stamps.data(), sizeof(stamps[0]), VK_QUERY_RESULT_64_BIT));
1149 const std::uint64_t mask = unlimited >> (64 - ctx.timestampBits);
1150 const auto milliseconds = [this, mask](std::uint64_t before, std::uint64_t after) {
1151 return static_cast<double>((after - before) & mask) *
1152 ctx.properties.limits.timestampPeriod / 1'000'000.0;
1153 };
1154 lastTiming = {milliseconds(stamps[0], stamps[1]), milliseconds(stamps[0], stamps[2])};
1155}
1156
1157bool VulkanGame::Impl::draw(const DrawFrame& params)
1158{
1159 // Two frames in flight. The slot's fence waits until the GPU is done with that slot's buffers.
1160 if (params.ui.size() > maxQuads)
1161 {
1162 throw std::runtime_error("The pixel UI exceeded its quad budget.");
1163 }
1164 auto& slot = slots[frame % framesInFlight];
1165 const auto fence = slot.fence.get();
1166 check(vkWaitForFences(ctx.device, 1, &fence, VK_TRUE, unlimited));
1167 readTiming(slot, (frame % framesInFlight) * 3);
1168
1169 std::uint32_t index{};
1170 const auto acquired = vkAcquireNextImageKHR(ctx.device, swap.handle.get(), unlimited,
1171 slot.available.get(), {}, &index);
1172 if (acquired == VK_ERROR_OUT_OF_DATE_KHR)
1173 {
1174 outdated = true;
1175 return false;
1176 }
1177 if (acquired != VK_SUBOPTIMAL_KHR)
1178 {
1179 check(acquired);
1180 }
1181 Buffer capture;
1182 if (!params.capture.empty())
1183 {
1184 if (!swap.captureSupported)
1185 {
1186 throw std::runtime_error("This swapchain cannot be captured.");
1187 }
1188 capture.create(ctx, static_cast<VkDeviceSize>(swap.extent.width) * swap.extent.height * 4,
1189 VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
1190 }
1191
1192 // Quad is trivially copyable. size_bytes() is the span's extent times the element size.
1193 std::memcpy(slot.ui.mapped, params.ui.data(), params.ui.size_bytes());
1194 auto cmd = slot.commands;
1195 startCommands(cmd);
1196 const unsigned first = (frame % framesInFlight) * 3;
1197 if (queries.get())
1198 {
1199 vkCmdResetQueryPool(cmd, queries.get(), first, 3);
1200 vkCmdWriteTimestamp2(cmd, VK_PIPELINE_STAGE_2_TOP_OF_PIPE_BIT, queries.get(), first);
1201 }
1202 if (params.stroke || params.ticks)
1203 {
1204 beginCompute(cmd);
1205 if (params.stroke)
1206 {
1207 recordBrush(cmd, *params.stroke);
1208 }
1209 recordSim(cmd, params.ticks);
1210 }
1211 if (queries.get())
1212 {
1213 vkCmdWriteTimestamp2(cmd, computeStage, queries.get(), first + 1);
1214 }
