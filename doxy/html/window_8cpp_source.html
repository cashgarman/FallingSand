15Window::Window(bool show)
16{
17 // Per-monitor DPI so client pixels match the swapchain extent.
18 SetProcessDpiAwarenessContext(DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2);
19
20 // Module instance used when registering the class and creating the HWND.
21 HINSTANCE instance = GetModuleHandleW(nullptr);
22
23 // Value-initialize the window class so unused Win32 fields stay zero.
24 WNDCLASSW type{};
25 type.lpfnWndProc = route;
26 type.hInstance = instance;
27 type.hCursor = LoadCursorW(nullptr, IDC_ARROW);
28 type.lpszClassName = className;
29 type.hIcon = LoadIconW(nullptr, IDI_APPLICATION);
30 if (!RegisterClassW(&type))
31 {
32 // Failure here is not a value the caller can recover. An exception leaves
33 // no Window object, so the destructor does not run for this attempt.
34 throw std::runtime_error("Could not register the Falling Sand window.");
35 }
36
37 // Default client size before converting to outer window dimensions.
38 RECT bounds{0, 0, 1280, 800};
39 AdjustWindowRectExForDpi(&bounds, windowStyle, FALSE, 0, GetDpiForSystem());
40
41 // Create the HWND and bind it to this Window through WM_NCCREATE.
42 handle_ = CreateWindowExW(0, className, L"Falling Sand", windowStyle, CW_USEDEFAULT,
43 CW_USEDEFAULT, bounds.right - bounds.left, bounds.bottom - bounds.top,
44 nullptr, nullptr, instance, this);
45 if (!handle_)
46 {
47 // Undo class registration when CreateWindowExW fails.
48 UnregisterClassW(className, instance);
49
50 throw std::runtime_error("Could not create the Falling Sand window.");
51 }
52
53 // Headless benchmark runs keep the HWND hidden.
54 if (show)
55 {
56 ShowWindow(handle_, SW_SHOW);
57 }
58}

87bool Window::poll()
88{
89 // Drain the thread queue. WM_QUIT ends the frame loop.
90 MSG msg{};
91 while (PeekMessageW(&msg, nullptr, 0, 0, PM_REMOVE))
92 {
93 if (msg.message == WM_QUIT)
94 {
95 return false;
96 }
97
98 // Route keyboard state through TranslateMessage before dispatch.
99 TranslateMessage(&msg);
100 DispatchMessageW(&msg);
101 }
102 return running_;
103}

132LRESULT CALLBACK Window::route(HWND handle, UINT message, WPARAM wparam, LPARAM lparam)
133{
134 if (message == WM_NCCREATE)
135 {
136 // lparam is the CREATESTRUCT for this message. The pointer stored in
137 // lpCreateParams is the Window* passed to CreateWindowExW.
138 const auto* create = reinterpret_cast<const CREATESTRUCTW*>(lparam);
139 auto* window = static_cast<Window*>(create->lpCreateParams);
140 window->handle_ = handle;
141
142 // Store the Window* as an integer-sized slot, then read that same pointer back.
143 SetWindowLongPtrW(handle, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(window));
144 }
145
146 // Resolve the Window instance stored during WM_NCCREATE.
147 auto* window = reinterpret_cast<Window*>(GetWindowLongPtrW(handle, GWLP_USERDATA));
148
149 // Before WM_NCCREATE finishes, forward to the default window procedure.
150 return window ? window->process(message, wparam, lparam)
151 : DefWindowProcW(handle, message, wparam, lparam);
152}

154LRESULT Window::process(UINT message, WPARAM wparam, LPARAM lparam)
155{
156 // The messages this window understands. Anything else goes to DefWindowProc.
157 switch (message)
158 {
159 case WM_CLOSE:
160 // User close stops the frame loop; WM_DESTROY follows from DefWindowProc.
161 running_ = false;
162 return 0;
163 case WM_DESTROY:
164 // Posted quit ends poll() on the next WM_QUIT.
165 PostQuitMessage(0);
166 return 0;
167 case WM_GETMINMAXINFO: {
168 // Keep the sandbox HUD usable on small displays.
169 auto* info = reinterpret_cast<MINMAXINFO*>(lparam);
170 info->ptMinTrackSize = {930, 620};
171 return 0;
172 }
173 case WM_DPICHANGED: {
174 // Apply the suggested frame when DPI changes on this monitor.
175 const auto& r = *reinterpret_cast<const RECT*>(lparam);
176 SetWindowPos(handle_, nullptr, r.left, r.top, r.right - r.left, r.bottom - r.top,
177 SWP_NOZORDER | SWP_NOACTIVATE);
178 return 0;
179 }
180 case WM_KEYDOWN:
181
182 // Bit 30 of lparam is the previous key state. A set bit means this is a repeat.
183 if (wparam < input_.pressed.size() && (lparam & (1L << 30)) == 0)
184 {
185 input_.pressed[wparam] = true;
186 }
187 return 0;
188 case WM_MOUSEWHEEL:
189 // Accumulate wheel delta so partial notches count toward one step.
190 wheelRemainder_ += GET_WHEEL_DELTA_WPARAM(wparam);
191 input_.wheel += wheelRemainder_ / WHEEL_DELTA;
192 wheelRemainder_ %= WHEEL_DELTA;
193 return 0;
194 case WM_MOUSEMOVE: {
195 // GET_X_LPARAM sign-extends the low word. A narrowing cast to float is explicit.
196 input_.mouse = {static_cast<float>(GET_X_LPARAM(lparam)),
197 static_cast<float>(GET_Y_LPARAM(lparam))};
198 TRACKMOUSEEVENT track{sizeof(track), TME_LEAVE, handle_, 0};
199 TrackMouseEvent(&track);
200 return 0;
201 }
202 case WM_MOUSELEAVE:
203 // Ignore leave events while capture still owns the pointer for a drag.
204 if (GetCapture() != handle_)
205 {
206 input_.mouse = {-1, -1};
207 }
208 return 0;
209 case WM_LBUTTONDOWN:
210 case WM_RBUTTONDOWN:
211 case WM_MBUTTONDOWN:
212 // Latch paint, erase, or stir while capture keeps tracking outside the client.
213 input_.mouse = {static_cast<float>(GET_X_LPARAM(lparam)),
214 static_cast<float>(GET_Y_LPARAM(lparam))};
215 input_.paint |= message == WM_LBUTTONDOWN;
216 input_.erase |= message == WM_RBUTTONDOWN;
217 input_.stir |= message == WM_MBUTTONDOWN;
218 if (message == WM_LBUTTONDOWN)
219 {
220 input_.paintDown = input_.mouse;
221 input_.paintPressed = true;
222 }
223 input_.erasePressed |= message == WM_RBUTTONDOWN;
224 input_.stirPressed |= message == WM_MBUTTONDOWN;
225 SetCapture(handle_);
226 return 0;
227 case WM_LBUTTONUP:
228 case WM_RBUTTONUP:
229 case WM_MBUTTONUP:
230 // Clear the matching button and release capture when none remain down.
231 input_.paint &= message != WM_LBUTTONUP;
232 input_.erase &= message != WM_RBUTTONUP;
233 input_.stir &= message != WM_MBUTTONUP;
234 if (!input_.paint && !input_.erase && !input_.stir)
235 {
236 ReleaseCapture();
237 }
238 return 0;
239 case WM_CAPTURECHANGED:
240 case WM_KILLFOCUS:
241 // Dropping focus or capture cancels in-progress drags.
242 input_.paint = input_.erase = input_.stir = false;
243 return 0;
244 default:
245 return DefWindowProcW(handle_, message, wparam, lparam);
246 }
